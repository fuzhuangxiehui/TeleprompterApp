 135 |     fun openFrontCamera() {
 136 |         cameraManager = context.getSystemService(Context.CAMERA_SERVICE) as CameraManager
 137 | 
 138 |         try {
 139 |             // Find front camera
 140 |             frontCameraId = null
 141 |             for (id in cameraManager!!.cameraIdList) {
 142 |                 val chars = cameraManager!!.getCameraCharacteristics(id)
 143 |                 if (chars.get(CameraCharacteristics.LENS_FACING) == CameraCharacteristics.LENS_FACING_FRONT) {
 144 |                     frontCameraId = id
 145 |                     break
 146 |                 }
 147 |             }
 148 | 
 149 |             if (frontCameraId == null) {
 150 |                 Log.e(TAG, "No front camera found on this device")
 151 |                 Handler(context.mainLooper).post {
 152 |                     android.widget.Toast.makeText(context, "未找到前置摄像头", android.widget.Toast.LENGTH_LONG).show()
 153 |                 }
 154 |                 return
 155 |             }
 156 | 
 157 |             // Choose a supported video size (sensor reports landscape dimensions)
 158 |             val characteristics = cameraManager!!.getCameraCharacteristics(frontCameraId!!)
 159 |             val map = characteristics.get(CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP)
 160 |             val videoSizes = map?.getOutputSizes(MediaRecorder::class.java)
 161 | 
 162 |             // Read zoom range and active array for crop-region zoom
 163 |             activeArrayRect = characteristics.get(CameraCharacteristics.SENSOR_INFO_ACTIVE_ARRAY_SIZE)
 164 |             val maxDigitalZoom = characteristics.get(CameraCharacteristics.SCALER_AVAILABLE_MAX_DIGITAL_ZOOM)
 165 |                 ?: 1.0f
 166 |             maxZoomRatio = maxDigitalZoom.coerceAtMost(5.0f)  // cap at 5x for safety
 167 |             Log.d(TAG, "Active array: ${activeArrayRect}, max digital zoom: ${maxDigitalZoom}")
 168 | 
 169 |             if (!videoSizes.isNullOrEmpty()) {
 170 |                 // Prefer 1920x1080 (landscape sensor dimensions)
 171 |                 val preferred = videoSizes.firstOrNull {
 172 |                     it.width == 1920 && it.height == 1080
 173 |                 }
 174 |                 if (preferred != null) {
 175 |                     sensorWidth = preferred.width   // 1920
 176 |                     sensorHeight = preferred.height  // 1080
 177 |                 } else {
 178 |                     // Pick largest size that is at most 1920x1080
 179 |                     val sorted = videoSizes.sortedByDescending { it.width * it.height }
 180 |                     val chosen = sorted.firstOrNull { it.width <= 1920 && it.height <= 1080 }
 181 |                         ?: sorted.last()
 182 |                     sensorWidth = chosen.width
 183 |                     sensorHeight = chosen.height
 184 |                 }
 185 |                 Log.d(TAG, "Chosen sensor size: ${sensorWidth}x${sensorHeight} (portrait: ${portraitWidth}x${portraitHeight})")
 186 |                 Log.d(TAG, "Available sensor sizes: ${videoSizes.map { "${it.width}x${it.height}" }}")
 187 |             }
 188 | 
 189 |             // Check permission
 190 |             if (context.checkSelfPermission(android.Manifest.permission.CAMERA) !=
 191 |                 android.content.pm.PackageManager.PERMISSION_GRANTED
 192 |             ) {
 193 |                 Log.e(TAG, "Camera permission not granted")
 194 |                 return
 195 |             }
 196 | 
 197 |             openCameraDevice()
 198 | 
 199 |         } catch (e: Exception) {
 200 |             Log.e(TAG, "Failed to open front camera", e)
 201 |             Handler(context.mainLooper).post {
 202 |                 android.widget.Toast.makeText(context, "打开摄像头失败: ${e.message}", android.widget.Toast.LENGTH_LONG).show()
 203 |             }
 204 |         }
 205 |     }

 239 |     private fun startPreview() {
 240 |         val device = cameraDevice ?: return
 241 |         val surfaceTexture = textureView.surfaceTexture
 242 |         if (surfaceTexture == null) {
 243 |             Log.w(TAG, "SurfaceTexture not ready")
 244 |             return
 245 |         }
 246 |         try {
 247 |             // Set buffer size to match sensor dimensions for correct aspect ratio
 248 |             surfaceTexture.setDefaultBufferSize(sensorWidth, sensorHeight)
 249 | 
 250 |             val surface = Surface(surfaceTexture)
 251 | 
 252 |             val requestBuilder = device.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW).apply {
 253 |                 addTarget(surface)
 254 |                 // Apply zoom to reduce wide-angle distortion
 255 |                 if (zoomRatio > 1.0f) {
 256 |                     set(CaptureRequest.SCALER_CROP_REGION, getCropRegion())
 257 |                 }
 258 |             }
 259 | 
 260 |             device.createCaptureSession(
 261 |                 listOf(surface),
 262 |                 object : CameraCaptureSession.StateCallback() {
 263 |                     override fun onConfigured(session: CameraCaptureSession) {
 264 |                         Log.d(TAG, "Preview session configured OK")
 265 |                         captureSession = session
 266 |                         try {
 267 |                             requestBuilder.build().let { request ->
 268 |                                 session.setRepeatingRequest(request, null, backgroundHandler)
 269 |                             }
 270 |                         } catch (e: Exception) {
 271 |                             Log.e(TAG, "Failed to start repeating preview", e)
 272 |                         }
 273 |                     }
 274 | 
 275 |                     override fun onConfigureFailed(session: CameraCaptureSession) {
 276 |                         Log.e(TAG, "Preview session configuration FAILED")
 277 |                     }
 278 |                 },

 293 |      * Set zoom ratio. 1.0 = no zoom (wide angle), 1.5 = moderate zoom.

 298 |         zoomRatio = ratio.coerceIn(1.0f, maxZoomRatio)
 299 |         Log.d(TAG, "Zoom set to ${zoomRatio}x")
 300 |         // Restart preview to apply new zoom
 301 |         if (isOpened && !isRecording) {
 302 |             startPreview()
 303 |         }
 304 |     }

 306 |     fun getZoom(): Float = zoomRatio
 307 |     fun getMaxZoom(): Float = maxZoomRatio

 309 |     /**
 310 |      * Calculate the crop region for the current zoom ratio.
 311 |      * Zoom works by cropping the sensor active array — a smaller crop = more zoom.
 312 |      * This is equivalent to increasing focal length (reducing wide-angle distortion).
 313 |      */
 314 |     private fun getCropRegion(): android.graphics.Rect {
 315 |         val array = activeArrayRect ?: return android.graphics.Rect(0, 0, sensorWidth, sensorHeight)
 316 |         if (zoomRatio <= 1.0f) return android.graphics.Rect(array)
 317 | 
 318 |         val w = (array.width() / zoomRatio).toInt()
 319 |         val h = (array.height() / zoomRatio).toInt()
 320 |         val left = (array.width() - w) / 2 + array.left
 321 |         val top = (array.height() - h) / 2 + array.top
 322 |         return android.graphics.Rect(left, top, left + w, top + h)
 323 |     }

 325 |     fun startRecording(outputPath: String): Boolean {
 326 |         if (cameraDevice == null || !isOpened) {
 327 |             Log.e(TAG, "Camera not opened, cannot record")
 328 |             Handler(context.mainLooper).post {
 329 |                 android.widget.Toast.makeText(context, "摄像头未就绪", android.widget.Toast.LENGTH_SHORT).show()
 330 |             }
 331 |             return false
 332 |         }
 333 | 
 334 |         if (isRecording) {
 335 |             Log.w(TAG, "Already recording")
 336 |             return false
 337 |         }
 338 | 
 339 |         try {
 340 |             // Close existing preview session
 341 |             captureSession?.close()
 342 |             captureSession = null
 343 | 
 344 |             // Setup MediaRecorder
 345 |             val recorder = MediaRecorder()
 346 |             mediaRecorder = recorder
 347 | 
 348 |             recorder.setAudioSource(MediaRecorder.AudioSource.MIC)
 349 |             recorder.setVideoSource(MediaRecorder.VideoSource.SURFACE)
 350 |             recorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
 351 |             recorder.setOutputFile(outputPath)
 352 |             recorder.setVideoEncodingBitRate(8 * 1024 * 1024) // 8 Mbps for 1080p
 353 |             recorder.setVideoFrameRate(30)
 354 |             // Must use sensor dimensions (landscape) for setVideoSize
 355 |             // The orientation hint tells the player to display as portrait
 356 |             recorder.setVideoSize(sensorWidth, sensorHeight)
 357 |             recorder.setVideoEncoder(MediaRecorder.VideoEncoder.H264)
 358 |             recorder.setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
 359 |             recorder.setAudioEncodingBitRate(128000)
 360 |             recorder.setAudioSamplingRate(44100)
 361 |             // Front camera portrait = 270 degrees
 362 |             recorder.setOrientationHint(270)
 363 | 
 364 |             recorder.prepare()
 365 |             Log.d(TAG, "MediaRecorder prepared, sensor: ${sensorWidth}x${sensorHeight}, portrait hint: 270°")
 366 | 
 367 |             val surfaceTexture = textureView.surfaceTexture!!
 368 |             surfaceTexture.setDefaultBufferSize(sensorWidth, sensorHeight)
 369 |             val previewSurface = Surface(surfaceTexture)
 370 |             val recorderSurface = recorder.surface
 371 | 
 372 |             // Create capture session with both surfaces
 373 |             val captureRequestBuilder = cameraDevice!!.createCaptureRequest(CameraDevice.TEMPLATE_RECORD).apply {
 374 |                 addTarget(previewSurface)
 375 |                 addTarget(recorderSurface)
 376 |                 // Apply zoom during recording too
 377 |                 if (zoomRatio > 1.0f) {
 378 |                     set(CaptureRequest.SCALER_CROP_REGION, getCropRegion())
 379 |                 }
 380 |             }
 381 | 
 382 |             cameraDevice?.createCaptureSession(
 383 |                 listOf(previewSurface, recorderSurface),
 384 |                 object : CameraCaptureSession.StateCallback() {
 385 |                     override fun onConfigured(session: CameraCaptureSession) {
 386 |                         captureSession = session
 387 |                         try {
 388 |                             captureRequestBuilder.build().let { request ->
 389 |                                 session.setRepeatingRequest(request, null, backgroundHandler)
 390 |                             }
 391 |                             // Start recording
 392 |                             Handler(context.mainLooper).post {
 393 |                                 try {
 394 |                                     recorder.start()
 395 |                                     isRecording = true
 396 |                                     currentVideoPath = outputPath
 397 |                                     Log.d(TAG, "Recording started successfully: $outputPath")
 398 |                                 } catch (e: Exception) {
 399 |                                     Log.e(TAG, "mediaRecorder.start() failed", e)
 400 |                                     Handler(context.mainLooper).post {
 401 |                                         android.widget.Toast.makeText(context, "录制启动失败: ${e.message}", android.widget.Toast.LENGTH_LONG).show()
 402 |                                     }
 403 |                                 }
 404 |                             }
 405 |                         } catch (e: Exception) {
 406 |                             Log.e(TAG, "Failed to set repeating request for recording", e)
 407 |                         }
 408 |                     }
 409 | 
 410 |                     override fun onConfigureFailed(session: CameraCaptureSession) {
 411 |                         Log.e(TAG, "Recording session configuration FAILED")
 412 |                         Handler(context.mainLooper).post {
 413 |                             android.widget.Toast.makeText(context, "录制会话配置失败", android.widget.Toast.LENGTH_LONG).show()
 414 |                         }
 415 |                     }
 416 |                 },
 417 |                 backgroundHandler
 418 |             )
 419 | 
 420 |             return true
 421 | 
 422 |         } catch (e: Exception) {
 423 |             Log.e(TAG, "Error starting recording", e)
 424 |             Handler(context.mainLooper).post {
 425 |                 android.widget.Toast.makeText(context, "录制启动异常: ${e.message}", android.widget.Toast.LENGTH_LONG).show()
 426 |             }
 427 |             return false
 428 |         }
 429 |     }

 431 |     fun stopRecording() {
 432 |         if (!isRecording) return
 433 |         isRecording = false
 434 | 
 435 |         val savedPath = currentVideoPath
 436 | 
 437 |         try {
 438 |             mediaRecorder?.apply {
 439 |                 stop()
 440 |                 release()
 441 |             }
 442 |             mediaRecorder = null
 443 |         } catch (e: Exception) {
 444 |             Log.e(TAG, "Error stopping mediaRecorder", e)
 445 |             try {
 446 |                 mediaRecorder?.release()
 447 |                 mediaRecorder = null
 448 |             } catch (_: Exception) {}
 449 |         }
 450 | 
 451 |         // Save to gallery so it's visible in album
 452 |         if (savedPath != null) {
 453 |             saveVideoToGallery(savedPath)
 454 |         }
 455 | 
 456 |         // Restart preview
 457 |         try {
 458 |             startPreview()
 459 |         } catch (e: Exception) {
 460 |             Log.e(TAG, "Error restarting preview after recording", e)
 461 |         }
 462 | 
 463 |         Log.d(TAG, "Recording stopped. File: $savedPath")
 464 | 
 465 |         // Notify callback
 466 |         onRecordingStopped?.invoke(savedPath)
 467 |     }

 469 |     private fun saveVideoToGallery(filePath: String) {
 470 |         try {
 471 |             val file = java.io.File(filePath)
 472 |             if (!file.exists()) {
 473 |                 Log.e(TAG, "Video file does not exist: $filePath")
 474 |                 return
 475 |             }
 476 | 
 477 |             val values = ContentValues().apply {
 478 |                 put(MediaStore.Video.Media.TITLE, file.name)
 479 |                 put(MediaStore.Video.Media.DISPLAY_NAME, file.name)
 480 |                 put(MediaStore.Video.Media.MIME_TYPE, "video/mp4")
 481 |                 put(MediaStore.Video.Media.DATE_ADDED, System.currentTimeMillis() / 1000)
 482 |                 put(MediaStore.Video.Media.DATE_TAKEN, System.currentTimeMillis())
 483 |                 put(MediaStore.Video.Media.SIZE, file.length())
 484 |             }
 485 | 
 486 |             val uri = context.contentResolver.insert(MediaStore.Video.Media.EXTERNAL_CONTENT_URI, values)
 487 |             if (uri != null) {
 488 |                 context.contentResolver.openOutputStream(uri)?.use { output ->
 489 |                     java.io.FileInputStream(file).use { input ->
 490 |                         input.copyTo(output)
 491 |                     }
 492 |                 }
 493 |                 Log.d(TAG, "Video saved to gallery: $uri")
 494 |                 Handler(context.mainLooper).post {
 495 |                     android.widget.Toast.makeText(context, "视频已保存到相册", android.widget.Toast.LENGTH_LONG).show()
 496 |                 }
 497 |             } else {
 498 |                 Log.e(TAG, "Failed to insert video into MediaStore")
 499 |             }
 500 |         } catch (e: Exception) {
 501 |             Log.e(TAG, "Error saving video to gallery", e)
 502 |             Handler(context.mainLooper).post {
 503 |                 android.widget.Toast.makeText(context, "视频保存到相册失败，文件在: $filePath", android.widget.Toast.LENGTH_LONG).show()
 504 |             }
 505 |         }
 506 |     }

 676 |     private fun startCamera() {
 677 |         if (cameraHelper != null) {
 678 |             cameraHelper?.close()
 679 |         }
 680 |         cameraHelper = CameraHelper(this, cameraPreview)
 681 | 
 682 |         // Set default zoom to reduce wide-angle distortion
 683 |         // Will be applied when preview starts
 684 |         currentZoom = 1.4f
 685 | 
 686 |         // Adjust TextureView aspect ratio to match camera preview
 687 |         // This prevents the preview from being stretched/distorted
 688 |         adjustPreviewAspectRatio()
 689 | 
 690 |         if (cameraPreview.isAvailable) {
 691 |             // Surface already ready, open camera immediately
 692 |             cameraHelper?.openFrontCamera()
 693 |             // Apply zoom after camera opens (delayed to ensure camera is ready)
 694 |             cameraPreview.postDelayed({
 695 |                 cameraHelper?.setZoom(currentZoom)
 696 |             }, 500)
 697 |         } else {
 698 |             // Wait for surface
 699 |             cameraPreview.surfaceTextureListener = object : TextureView.SurfaceTextureListener {
 700 |                 override fun onSurfaceTextureAvailable(surface: android.graphics.SurfaceTexture, width: Int, height: Int) {
 701 |                     cameraHelper?.openFrontCamera()
 702 |                     cameraPreview.postDelayed({
 703 |                         cameraHelper?.setZoom(currentZoom)
 704 |                     }, 500)
 705 |                 }
 706 |                 override fun onSurfaceTextureSizeChanged(surface: android.graphics.SurfaceTexture, width: Int, height: Int) {
 707 |                     cameraHelper?.updatePreview()
 708 |                     adjustPreviewAspectRatio()
 709 |                 }
 710 |                 override fun onSurfaceTextureDestroyed(surface: android.graphics.SurfaceTexture): Boolean {
 711 |                     cameraHelper?.close()
 712 |                     return true
 713 |                 }
 714 |                 override fun onSurfaceTextureUpdated(surface: android.graphics.SurfaceTexture) {}
 715 |             }
 716 |         }

 718 | 
 719 |     /**
 720 |      * Adjust the TextureView's dimensions to match the camera's portrait aspect ratio.
 721 |      * Without this, the full-screen TextureView stretches the camera preview to fill
 722 |      * the screen, causing the person to appear stretched/distorted.
 723 |      * 
 724 |      * We fit the preview to fill the screen width, and adjust the height to match
 725 |      * the camera's portrait aspect ratio. If the calculated height is taller than the
 726 |      * screen, we fill the screen entirely (camera aspect ratio is very close to screen ratio,
 727 |      * so any slight mismatch is negligible). The key is ensuring the buffer ratio matches.
 728 |      */
 729 |     private fun adjustPreviewAspectRatio() {
 730 |         val helper = cameraHelper ?: return
 731 |         val portraitW = helper.portraitWidth
 732 |         val portraitH = helper.portraitHeight
 733 |         if (portraitW <= 0 || portraitH <= 0) return
 734 | 
 735 |         cameraPreview.post {
 736 |             val container = cameraPreview.parent as? android.view.ViewGroup ?: return@post
 737 |             val containerW = container.width
 738 |             val containerH = container.height
 739 |             if (containerW <= 0 || containerH <= 0) return@post
 740 | 
 741 |             // Camera portrait aspect ratio (e.g. 1080:1920 = 9:16)
 742 |             val cameraRatio = portraitW.toFloat() / portraitH.toFloat()
 743 |             // Container aspect ratio
 744 |             val containerRatio = containerW.toFloat() / containerH.toFloat()
 745 | 
 746 |             val params = cameraPreview.layoutParams
 747 |             if (abs(containerRatio - cameraRatio) < 0.01f) {
 748 |                 // Ratios match closely enough, use match_parent
 749 |                 // No adjustment needed
 750 |             } else {
 751 |                 // Fit width to container, calculate height from camera ratio
 752 |                 val previewW = containerW
 753 |                 val previewH = (containerW / cameraRatio).toInt()
 754 |                 params.width = previewW
 755 |                 params.height = previewH.coerceAtMost(containerH)
 756 |                 cameraPreview.layoutParams = params
 757 |             }
 758 |         }
 759 |     }
